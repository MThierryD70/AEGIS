cmake_minimum_required(VERSION 3.15)
project(aegis_cpp)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Trouve Python
find_package(Python REQUIRED COMPONENTS Interpreter Development)

# pybind11 — chemin récupéré depuis Python
execute_process(
    COMMAND "${Python_EXECUTABLE}" -c
        "import pybind11; print(pybind11.get_cmake_dir())"
    OUTPUT_VARIABLE PYBIND11_CMAKE_DIR
    OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE PYBIND11_RESULT
)

if(NOT PYBIND11_RESULT EQUAL 0)
    message(FATAL_ERROR
        "pybind11 introuvable. Lancez : pip install pybind11")
endif()

list(APPEND CMAKE_PREFIX_PATH "${PYBIND11_CMAKE_DIR}")
find_package(pybind11 REQUIRED CONFIG)

# OpenSSL - find_package cherche automatiquement
# Les chemins peuvent être passés via -DOPENSSL_INCLUDE_DIR etc.
find_package(OpenSSL REQUIRED)

if(NOT OpenSSL_FOUND)
    message(FATAL_ERROR
        "OpenSSL introuvable. Installez OpenSSL et relancez.")
endif()

message(STATUS "OpenSSL include : ${OPENSSL_INCLUDE_DIR}")
message(STATUS "OpenSSL libs    : ${OPENSSL_LIBRARIES}")

# Module Python
pybind11_add_module(aegis_cpp src/aegis_module.cpp)

target_include_directories(aegis_cpp PRIVATE
    ${OPENSSL_INCLUDE_DIR}
)

target_link_libraries(aegis_cpp PRIVATE
    OpenSSL::SSL
    OpenSSL::Crypto
)

# Copie le .pyd dans cpp/bin/
add_custom_command(TARGET aegis_cpp POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory
        "${CMAKE_SOURCE_DIR}/bin"
    COMMAND ${CMAKE_COMMAND} -E copy
        $<TARGET_FILE:aegis_cpp>
        "${CMAKE_SOURCE_DIR}/bin/$<TARGET_FILE_NAME:aegis_cpp>"
)
